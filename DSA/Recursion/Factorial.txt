

// Find Factorial 


using System;

public class Recursion
{
    public static int fact(int n)
    {
        // Base case
        if (n == 0)
        {
            return 1;
        }

        // Recursive call
        int fnm1 = fact(n - 1);

        int fn = n * fnm1;

        return fn;
    }

    public static void Main(string[] args)
    {
        int n = 5;

        int result = fact(n);

        Console.WriteLine(result);
    }
}
