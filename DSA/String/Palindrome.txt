
using System;

public class HelloWorld {
    public static void Main(string[] args) {

        string str = "madam";

        for(int i=0; i<str.Length/2; i++){

            if(str[i] != str[str.Length-1-i]){
                
                Console.WriteLine(false);
                return;
            }
        }

        Console.WriteLine(true);
    }
}
