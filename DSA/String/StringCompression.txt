using System;

public class HelloWorld
{
    public static void Main(string[] args)
    {
        string str = "aaabbcccdd";

        string newStr = "";

        for (int i = 0; i < str.Length; i++)
        {
            int count = 1;

            while (i < str.Length - 1 && str[i] == str[i + 1])
            {
                count++;
                i++;
            }

            newStr += str[i];

            if (count > 1)
            {
                newStr += count.ToString();
            }
        }

        Console.WriteLine(newStr);
    }
}
