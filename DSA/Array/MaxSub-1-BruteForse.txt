
using System;

public class HelloWorld {
    public static void Main(string[] args) {

        int[] arr = {1,-2,6,-1,3};
        int currSum = 0;
        int maxSum = int.MinValue;

        for (int i = 0; i < arr.Length; i++)
        {
            int start = i;

            for (int j = i; j < arr.Length; j++)
            {
                int end = j;
                currSum = 0;
                for (int k = start; k <= end; k++)
                {
                   currSum += arr[k];
                }
                
                if(maxSum < currSum){
                    maxSum = currSum;
                }
            }

        }
        
        Console.WriteLine(maxSum);
        
    }
}

Output ==> 8;
