
using System;

public class HelloWorld {
    public static void Main(string[] args) {

        int[] arr = {-2, -3, 4, -1, -2, 1, 5, -3};
        int cs = 0;
        int ms = int.MinValue;

        for(int i=0; i<arr.Length; i++){

            cs = cs + arr[i];
            if(cs < 0){
                cs = 0;
            }
            ms = Math.Max(cs, ms);
        }
        Console.WriteLine(ms);
    }
}

Output ==> 7
