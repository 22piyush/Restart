

//Find Largest element and Push in End


using system;

public class BubbleSort
{
    public static void Main(string[] args)
    {
        int[] arr = { 5, 4, 1, 3, 2 };

        for(int i = 0; i < arr.Length - 1; i++)
        {
            int swap = 0;

            for(int j = 0; j < arr.Length - 1 - i; j++)
            {
                if(arr[j] > arr[j + 1])
                {
                    int temp = arr[j];
                    arr[j] = arr[j + 1];
                    arr[j + 1] = temp;

                    swap++;
                }
            }

            if(swap == 0)
            {
                break;
            }
        }

        foreach(int ar in arr)
        {
            Console.WriteLine(ar);
        }
    }
}
