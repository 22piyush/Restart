
Approach :- Get Element from Unsorted part and swap with sorted part

5,4,1,3,2
4,5,1,3,2
1,4,5,3,2
1,3,4,5,2
1,2,3,4,5


using system;

public class InsertionSort
{
    public static void Main(string[] args)
    {

        int[] arr = {5,4,1,3,2};

        for(int i=1; i<arr.Length; i++){

            int curr = arr[i];
            int prev = i-1;

            while(prev >= 0 && arr[prev] > curr){

                arr[prev+1] = arr[prev];
                prev--;  
            }

            arr[prev+1] = curr;

        }

        foreach(ar in arr){
            Console.WriteLine(ar);
        }
    }    
}
