// Reverse Each Word in a Sentence
const str = "I am learning JavaScript";

const words = str.split(" ");
let result = "";

for (let i = 0; i < words.length; i++) {

    let word = "";

    for (let j = words[i].length - 1; j >= 0; j--) {
        word += words[i][j];
    }

    result += word;

    if (i !== words.length - 1) {
        result += " ";
    }
}

console.log(result);
